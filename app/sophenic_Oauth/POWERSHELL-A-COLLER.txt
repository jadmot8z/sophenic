$ErrorActionPreference = "Stop"
$ProgressPreference = "SilentlyContinue"
Set-StrictMode -Version Latest

function Test-IsAdministrator {
    $Identity = [Security.Principal.WindowsIdentity]::GetCurrent()
    $Principal = New-Object Security.Principal.WindowsPrincipal($Identity)
    return $Principal.IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)
}

$TranscriptStarted = $false

try {
    Clear-Host
    Write-Host ""
    Write-Host "============================================================" -ForegroundColor Cyan
    Write-Host " SOPHENIC 5.4.0 - BUILD PROPRE DEPUIS LE ZIP" -ForegroundColor Cyan
    Write-Host "============================================================" -ForegroundColor Cyan
    Write-Host ""

    if ($env:OS -ne "Windows_NT") {
        throw "Ce code doit être exécuté sous Windows."
    }

    if (-not (Test-IsAdministrator)) {
        throw "Cette fenêtre n'est pas Administrateur. Ouvre PowerShell avec 'Exécuter en tant qu'administrateur'."
    }

    $Downloads = Join-Path $env:USERPROFILE "Downloads"
    $ZipPattern = "SOPHENIC-V5.4.0-AGENTIC-THINKING-PRO*.zip"
    $BuildDir = Join-Path $Downloads "SOPHENIC-5.4.0-BUILD"
    $FinalExe = Join-Path $Downloads "Sophenic Setup.exe"
    $LogFile = Join-Path $Downloads "Sophenic-build-5.4.0.log"

    Write-Host "[1/7] Recherche du ZIP..." -ForegroundColor Yellow
    $Zip = Get-ChildItem -LiteralPath $Downloads -File -Filter $ZipPattern -ErrorAction SilentlyContinue |
        Sort-Object LastWriteTime -Descending |
        Select-Object -First 1

    if (-not $Zip) {
        throw "ZIP introuvable dans Téléchargements. Nom attendu: $ZipPattern"
    }
    Write-Host "[OK] $($Zip.FullName)" -ForegroundColor Green

    Write-Host "[2/7] Extraction dans un dossier neuf..." -ForegroundColor Yellow
    if (Test-Path -LiteralPath $BuildDir) {
        Set-Location -LiteralPath $Downloads
        Remove-Item -LiteralPath $BuildDir -Recurse -Force
    }
    New-Item -ItemType Directory -Path $BuildDir -Force | Out-Null
    Expand-Archive -LiteralPath $Zip.FullName -DestinationPath $BuildDir -Force

    Write-Host "[3/7] Déblocage des fichiers Windows..." -ForegroundColor Yellow
    Get-ChildItem -LiteralPath $BuildDir -File -Recurse -ErrorAction SilentlyContinue |
        ForEach-Object {
            try { Unblock-File -LiteralPath $_.FullName -ErrorAction Stop } catch {}
        }

    Write-Host "[4/7] Recherche du projet SOPHENIC 5.4.0..." -ForegroundColor Yellow
    $Package = $null
    $Packages = Get-ChildItem -LiteralPath $BuildDir -Filter "package.json" -File -Recurse -ErrorAction SilentlyContinue |
        Where-Object { $_.FullName -notmatch "\\node_modules\\" }

    foreach ($Candidate in $Packages) {
        try {
            $Info = Get-Content -LiteralPath $Candidate.FullName -Raw | ConvertFrom-Json
            if ([string]$Info.name -eq "sophenic" -and [string]$Info.version -eq "5.4.0") {
                $Package = $Candidate
                break
            }
        } catch {}
    }

    if (-not $Package) {
        throw "package.json de SOPHENIC 5.4.0 introuvable après extraction."
    }

    $ProjectRoot = $Package.Directory.FullName
    $BuildScript = Join-Path $ProjectRoot "build-sophenic.ps1"

    Write-Host "[5/7] Vérification de build-sophenic.ps1..." -ForegroundColor Yellow
    if (-not (Test-Path -LiteralPath $BuildScript)) {
        $FoundScripts = Get-ChildItem -LiteralPath $BuildDir -Filter "*.ps1" -File -Recurse -ErrorAction SilentlyContinue |
            Select-Object -ExpandProperty FullName
        Write-Host "Scripts PS1 présents dans le ZIP :" -ForegroundColor Yellow
        $FoundScripts | ForEach-Object { Write-Host " - $_" -ForegroundColor DarkGray }
        throw "build-sophenic.ps1 est absent après extraction."
    }

    $BuildScriptInfo = Get-Item -LiteralPath $BuildScript
    if ($BuildScriptInfo.Length -lt 1000) {
        throw "build-sophenic.ps1 semble incomplet ($($BuildScriptInfo.Length) octets)."
    }
    Write-Host "[OK] $BuildScript" -ForegroundColor Green
    Write-Host "     Taille: $($BuildScriptInfo.Length) octets" -ForegroundColor DarkGray

    if (-not (Get-Command node.exe -ErrorAction SilentlyContinue)) { throw "Node.js est introuvable dans le PATH." }
    if (-not (Get-Command npm.cmd -ErrorAction SilentlyContinue)) { throw "npm est introuvable dans le PATH." }
    Write-Host "Node: $(node --version)" -ForegroundColor Cyan
    Write-Host "npm : $(npm --version)" -ForegroundColor Cyan

    if (Test-Path -LiteralPath $FinalExe) {
        Remove-Item -LiteralPath $FinalExe -Force
    }

    try {
        Start-Transcript -Path $LogFile -Force | Out-Null
        $TranscriptStarted = $true
    } catch {}

    Write-Host "[6/7] Tests + Next.js + Electron + NSIS..." -ForegroundColor Yellow
    Write-Host "Projet : $ProjectRoot" -ForegroundColor DarkGray
    Write-Host "Journal: $LogFile" -ForegroundColor DarkGray
    Set-Location -LiteralPath $ProjectRoot

    & powershell.exe -NoLogo -NoProfile -ExecutionPolicy Bypass -File "$BuildScript" -ElevatedBuild
    $BuildCode = $LASTEXITCODE
    if ($BuildCode -ne 0) {
        throw "Le build a échoué avec le code $BuildCode. Le journal reste dans $LogFile"
    }

    Write-Host "[7/7] Vérification de l'installateur..." -ForegroundColor Yellow
    if (-not (Test-Path -LiteralPath $FinalExe)) {
        $GeneratedExe = Get-ChildItem -LiteralPath $ProjectRoot -Filter "Sophenic Setup.exe" -File -Recurse -ErrorAction SilentlyContinue |
            Sort-Object LastWriteTime -Descending |
            Select-Object -First 1
        if ($GeneratedExe) {
            Copy-Item -LiteralPath $GeneratedExe.FullName -Destination $FinalExe -Force
        }
    }

    if (-not (Test-Path -LiteralPath $FinalExe)) {
        throw "Sophenic Setup.exe n'a pas été créé. Consulte $LogFile"
    }

    $ExeInfo = Get-Item -LiteralPath $FinalExe
    if ($ExeInfo.Length -lt 40000000) {
        throw "L'installateur semble incomplet: $([Math]::Round($ExeInfo.Length / 1MB, 1)) Mo."
    }

    $Stream = [System.IO.File]::OpenRead($FinalExe)
    try {
        $B0 = $Stream.ReadByte()
        $B1 = $Stream.ReadByte()
        if ($B0 -ne 0x4D -or $B1 -ne 0x5A) {
            throw "Le fichier final n'est pas un EXE Windows PE valide."
        }
    } finally {
        $Stream.Dispose()
    }

    Write-Host ""
    Write-Host "============================================================" -ForegroundColor Green
    Write-Host " SOPHENIC SETUP.EXE EST CREE" -ForegroundColor Green
    Write-Host "============================================================" -ForegroundColor Green
    Write-Host $FinalExe -ForegroundColor White
    Write-Host "Taille: $([Math]::Round($ExeInfo.Length / 1MB, 1)) Mo" -ForegroundColor Cyan
    Start-Process explorer.exe -ArgumentList "/select,`"$FinalExe`""
}
catch {
    Write-Host ""
    Write-Host "============================================================" -ForegroundColor Red
    Write-Host " ERREUR SOPHENIC" -ForegroundColor Red
    Write-Host "============================================================" -ForegroundColor Red
    Write-Host $_.Exception.Message -ForegroundColor Red
    Write-Host ""
    if (Get-Variable LogFile -ErrorAction SilentlyContinue) {
        Write-Host "Journal: $LogFile" -ForegroundColor Yellow
    }
}
finally {
    if ($TranscriptStarted) {
        try { Stop-Transcript | Out-Null } catch {}
    }
    Write-Host ""
    Write-Host "La fenêtre reste ouverte." -ForegroundColor Cyan
    Read-Host "Appuie sur ENTREE seulement quand tu as fini de lire"
}
